cmake_minimum_required(VERSION 3.28)
project(fifa96_loader C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Werror)
include_directories(include)
enable_testing()
add_library(fifa96_file src/fifa96_loader/fifa96_file.c)
add_executable(test_file tests/test_file.c)
target_link_libraries(test_file PRIVATE fifa96_file)
add_test(NAME test_file COMMAND test_file WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_executable(test_load_order tests/test_load_order.c)
add_test(NAME test_load_order COMMAND test_load_order WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_library(fifa96_tables src/fifa96_loader/fifa96_tables.c)
target_link_libraries(fifa96_tables PRIVATE fifa96_file)
add_executable(test_tables tests/test_tables.c)
target_link_libraries(test_tables PRIVATE fifa96_tables fifa96_file)
add_test(NAME test_tables COMMAND test_tables WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_library(fifa96_pog src/fifa96_loader/fifa96_pog.c)
add_executable(test_pog tests/test_pog.c)
target_link_libraries(test_pog PRIVATE fifa96_pog fifa96_file)
add_test(NAME test_pog COMMAND test_pog WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_library(fifa96_qfs src/fifa96_loader/fifa96_qfs.c)
add_executable(test_qfs tests/test_qfs.c)
target_link_libraries(test_qfs PRIVATE fifa96_qfs fifa96_file)
add_test(NAME test_qfs COMMAND test_qfs WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_library(fifa96_viv src/fifa96_loader/fifa96_viv.c)
add_library(fifa96_tgv src/fifa96_loader/fifa96_tgv.c)
add_executable(test_viv tests/test_viv.c)
target_link_libraries(test_viv PRIVATE fifa96_viv fifa96_file)
add_executable(test_tgv tests/test_tgv.c)
target_link_libraries(test_tgv PRIVATE fifa96_tgv fifa96_file)
add_test(NAME test_viv COMMAND test_viv WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_test(NAME test_tgv COMMAND test_tgv WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_executable(test_neg tests/test_neg.c)
target_link_libraries(test_neg PRIVATE fifa96_pog fifa96_qfs fifa96_tgv fifa96_viv fifa96_tables fifa96_file)
add_test(NAME test_neg COMMAND test_neg WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})

add_library(fifa96_envelope src/fifa96_loader/fifa96_envelope.c)
target_link_libraries(fifa96_envelope PRIVATE fifa96_file)
add_executable(test_envelope tests/test_envelope.c)

target_link_libraries(test_script PRIVATE fifa96_script fifa96_file)
add_test(NAME test_script COMMAND test_script WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})

target_link_libraries(fifa96_dump PRIVATE fifa96_envelope fifa96_qfs fifa96_pog fifa96_tgv fifa96_viv fifa96_file)
